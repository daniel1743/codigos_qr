// @vitest-environment happy-dom
/**
 * CRIPQER CA-11 · the REAL landing, mounted, with the Advisor inside it.
 *
 * These tests drive the landing the product actually ships: the same wrapper,
 * the same hook, the same large composer and the same conversational stages. The
 * only thing replaced is the transport at the far edge — the seam the brief
 * allows — so what is checked is the integration itself, not a copy of it.
 *
 * WHAT THEY PROVE, each named by the gate it serves:
 *  · typing reaches the canonical controller, and its PROCESSING/ANSWERED states
 *    are what the visitor sees;
 *  · three typed messages are three turns of ONE conversation (the transport's
 *    session reuse is proved separately, in `transport.test.ts`);
 *  · an authorised OPEN_HELP is INERT until a real click, and one click calls
 *    the injected router exactly once;
 *  · a second submit while a turn is open produces no second turn;
 *  · a failed turn leaves an explicit retry, and the retry appends a new turn
 *    without erasing the failure;
 *  · an out-of-contract payload becomes ERROR and its text never reaches the DOM;
 *  · the prototype agent's own sentence is NOT rendered for an Advisor turn.
 *
 * The idiom is the repository's own (`createRoot` + `act`, no Testing Library);
 * see CA-7.4-B's surface test for why.
 */

import { act, type ReactElement } from "react";
import { createRoot } from "react-dom/client";

import { afterEach, describe, expect, it, vi } from "vitest";

import { ApprovedChatLanding } from "../../index";
import type { AdvisorTransport } from "../../../../lib/conversion-advisor/ui/types";

(
  globalThis as typeof globalThis & { IS_REACT_ACT_ENVIRONMENT?: boolean }
).IS_REACT_ACT_ENVIRONMENT = true;

vi.setConfig({ testTimeout: 30_000 });

/* ============================== fixtures ================================ */

const ANSWER = "Tu página puede reunir tus enlaces y tu QR en un solo lugar.";
const SECOND = "Sí: puedes cambiarla cuando quieras.";
const THIRD = "El QR apunta siempre a tu página publicada.";

function response(message: string, actions: readonly unknown[] = []): unknown {
  return {
    version: 1,
    status: "ANSWERED",
    message,
    notice: null,
    actions,
    conversation: { turns_used: 1, turns_remaining: 2 },
  };
}

const OPEN_HELP = {
  kind: "NAVIGATE_INTERNAL",
  action_id: "OPEN_HELP",
  destination: "/help",
};

/** A transport the test releases by hand, so a turn can be seen in flight. */
function heldTransport(): {
  readonly transport: AdvisorTransport;
  readonly calls: string[];
  release: (value: unknown) => void;
  reject: (error: Error) => void;
} {
  const calls: string[] = [];
  let resolve: ((value: unknown) => void) | null = null;
  let fail: ((error: Error) => void) | null = null;
  return {
    calls,
    transport: ({ text }) => {
      calls.push(text);
      return new Promise<unknown>((res, rej) => {
        resolve = res;
        fail = rej;
      });
    },
    release: (value) => resolve?.(value),
    reject: (error) => fail?.(error),
  };
}

/* =============================== the DOM ================================ */

const mounted: Array<() => void> = [];

function mount(node: ReactElement): HTMLElement {
  const host = document.createElement("div");
  document.body.append(host);
  const root = createRoot(host);
  act(() => {
    root.render(node);
  });
  mounted.push(() => {
    act(() => {
      root.unmount();
    });
    host.remove();
  });
  return host;
}

async function settle(): Promise<void> {
  await act(async () => {
    await new Promise((resolve) => setTimeout(resolve, 0));
  });
}

/** The prototype agent's own delay; waited out so the input becomes usable. */
async function waitForLocalAgent(): Promise<void> {
  await act(async () => {
    await new Promise((resolve) => setTimeout(resolve, 1_250));
  });
}

function field(host: HTMLElement): HTMLTextAreaElement {
  // AnimatePresence keeps an exiting stage in the DOM, and framer-motion's exit
  // animation does not advance in happy-dom. The ACTIVE stage is the last one,
  // so both helpers take the last match instead of the first.
  const elements = host.querySelectorAll("#cripqer-composer");
  const element = elements[elements.length - 1];
  if (!(element instanceof HTMLTextAreaElement)) throw new Error("CA11_TEST: no composer");
  return element;
}

function type(host: HTMLElement, value: string): void {
  const element = field(host);
  const setter = Object.getOwnPropertyDescriptor(HTMLTextAreaElement.prototype, "value")?.set;
  if (setter === undefined) throw new Error("CA11_TEST: no native value setter");
  act(() => {
    setter.call(element, value);
    element.dispatchEvent(new Event("input", { bubbles: true }));
  });
}

function sendButton(host: HTMLElement): HTMLButtonElement {
  const buttons = host.querySelectorAll('button[aria-label="Enviar"]');
  const button = buttons[buttons.length - 1];
  if (!(button instanceof HTMLButtonElement)) throw new Error("CA11_TEST: no send button");
  return button;
}

function click(host: HTMLElement, selector: string): void {
  const element = host.querySelector(selector);
  if (!(element instanceof HTMLElement)) throw new Error(`CA11_TEST: no ${selector}`);
  act(() => {
    element.click();
  });
}

function statuses(host: HTMLElement): readonly (string | null)[] {
  return [...host.querySelectorAll("[data-status]")].map((node) =>
    node.getAttribute("data-status"),
  );
}

function advisorTexts(host: HTMLElement): readonly string[] {
  return [...host.querySelectorAll('[data-role="ADVISOR"] p')].map(
    (node) => node.textContent ?? "",
  );
}

async function sendTurn(host: HTMLElement, text: string): Promise<void> {
  type(host, text);
  act(() => {
    sendButton(host).click();
  });
  await settle();
}

afterEach(() => {
  while (mounted.length > 0) mounted.pop()?.();
  document.body.innerHTML = "";
});

/* ============================= the landing ============================== */

describe("CA-11 · the real landing mounts the Advisor", () => {
  it("shows the approved landing and performs no call when nothing is typed", () => {
    const held = heldTransport();
    const host = mount(<ApprovedChatLanding advisorTransport={held.transport} />);

    expect(host.textContent).toContain("Páginas increíbles");
    expect(field(host)).toBeTruthy();
    expect(held.calls).toEqual([]);
    expect(statuses(host)).toEqual([]);
  });

  it("sends the typed text and renders PROCESSING, then ANSWERED", async () => {
    const held = heldTransport();
    const host = mount(<ApprovedChatLanding advisorTransport={held.transport} />);

    await sendTurn(host, "hola");

    expect(held.calls).toEqual(["hola"]);
    expect(host.textContent).toContain("hola");
    expect(statuses(host)).toEqual(["PROCESSING"]);
    expect(sendButton(host).disabled).toBe(true);

    held.release(response(ANSWER));
    await settle();

    expect(statuses(host)).toEqual(["ANSWERED"]);
    expect(advisorTexts(host)).toEqual([ANSWER]);
    // The prototype agent's own greeting must not appear beside the Advisor's
    // validated answer — there is one assistant voice per turn.
    expect(host.textContent).not.toContain("¿Qué quieres crear hoy?");
  });

  it("continues ONE conversation across three typed turns", async () => {
    const calls: string[] = [];
    const transport: AdvisorTransport = async ({ text }) => {
      calls.push(text);
      return response([ANSWER, SECOND, THIRD][calls.length - 1] ?? ANSWER);
    };
    const host = mount(<ApprovedChatLanding advisorTransport={transport} />);

    await sendTurn(host, "uno");
    await waitForLocalAgent();
    await sendTurn(host, "dos");
    await waitForLocalAgent();
    await sendTurn(host, "tres");
    await settle();

    expect(calls).toEqual(["uno", "dos", "tres"]);
    expect(advisorTexts(host)).toEqual([ANSWER, SECOND, THIRD]);
    expect(statuses(host)).toEqual(["ANSWERED", "ANSWERED", "ANSWERED"]);
  });
});

/* ============================== OPEN_HELP =============================== */

describe("CA-11 · OPEN_HELP in the real product", () => {
  it("stays inert until a real click, then navigates exactly once", async () => {
    const onNavigate = vi.fn<(destination: "/help") => void>();
    const host = mount(
      <ApprovedChatLanding
        advisorTransport={async () => response(ANSWER, [OPEN_HELP])}
        onNavigate={onNavigate}
      />,
    );

    await sendTurn(host, "¿dónde está la ayuda?");
    await settle();
    await waitForLocalAgent();

    const button = host.querySelector('[data-action-id="OPEN_HELP"]');
    expect(button).toBeInstanceOf(HTMLButtonElement);
    expect((button as HTMLButtonElement).disabled).toBe(false);
    // Rendering, settling and an unrelated 1.25s of timers: still nothing.
    expect(onNavigate).not.toHaveBeenCalled();
    // The action's raw id is not visitor-facing text.
    expect(host.textContent).not.toContain("OPEN_HELP");

    click(host, '[data-action-id="OPEN_HELP"]');
    expect(onNavigate).toHaveBeenCalledTimes(1);
    expect(onNavigate).toHaveBeenCalledWith("/help");
  });

  it("leaves the button disabled when the host injects no router", async () => {
    const host = mount(
      <ApprovedChatLanding advisorTransport={async () => response(ANSWER, [OPEN_HELP])} />,
    );

    await sendTurn(host, "ayuda");
    await settle();

    const button = host.querySelector('[data-action-id="OPEN_HELP"]');
    expect(button).toBeInstanceOf(HTMLButtonElement);
    expect((button as HTMLButtonElement).disabled).toBe(true);
  });
});

/* ========================= concurrency and failure ====================== */

describe("CA-11 · the landing's failure and concurrency behaviour", () => {
  it("cannot produce a second turn from a second submit while one is open", async () => {
    const held = heldTransport();
    const host = mount(<ApprovedChatLanding advisorTransport={held.transport} />);

    type(host, "hola");
    act(() => {
      sendButton(host).click();
      // The DOM would be disabled after a commit, but a scripted second submit
      // bypasses that: the controller's single-flight rule is what must hold.
      sendButton(host).click();
    });
    await settle();

    expect(held.calls).toEqual(["hola"]);
    expect(statuses(host)).toEqual(["PROCESSING"]);
    held.release(response(ANSWER));
    await settle();
  });

  it("keeps a failed turn visible and recovers through an explicit retry", async () => {
    let call = 0;
    const transport: AdvisorTransport = async ({ text }) => {
      call += 1;
      if (call === 1) throw new Error("network down");
      expect(text).toBe("hola");
      return response(ANSWER);
    };
    const host = mount(<ApprovedChatLanding advisorTransport={transport} />);

    await sendTurn(host, "hola");
    await settle();
    expect(statuses(host)).toEqual(["ERROR"]);
    expect(host.textContent).toContain("Reintentar");
    // The composer recovers: the visitor can keep talking without a reload.
    await waitForLocalAgent();
    type(host, "otra pregunta");
    expect(sendButton(host).disabled).toBe(false);

    const retry = [...host.querySelectorAll("button")].find(
      (button) => button.textContent === "Reintentar",
    );
    expect(retry).toBeInstanceOf(HTMLButtonElement);
    act(() => {
      (retry as HTMLButtonElement).click();
    });
    await settle();

    expect(call).toBe(2);
    expect(statuses(host)).toEqual(["ERROR", "ANSWERED"]);
    expect(advisorTexts(host).at(-1)).toBe(ANSWER);
    // The failed attempt is still in the transcript, not rewritten as success.
    expect(statuses(host)[0]).toBe("ERROR");
  });

  it("renders an out-of-contract payload as ERROR and never its message", async () => {
    const hostile = {
      version: 1,
      status: "ANSWERED",
      message: "TEXTO QUE NO DEBE RENDERIZARSE",
      notice: null,
      actions: [{ kind: "NAVIGATE", href: "https://evil.invalid/" }],
      conversation: null,
    };
    const host = mount(<ApprovedChatLanding advisorTransport={async () => hostile} />);

    await sendTurn(host, "hola");
    await settle();

    expect(statuses(host)).toEqual(["ERROR"]);
    expect(host.textContent).not.toContain("TEXTO QUE NO DEBE RENDERIZARSE");
    expect(host.innerHTML).not.toContain("evil.invalid");
  });
});
